Завдання №9 — Варіант 4: Текстовий редактор

Умова (варіант 4): Реалізувати редактор, який дозволяє виконувати форматування тексту. Обов'язкові елементи форматування: напівжирність, нахил, підкреслювання, колір тексту, регістр символів. Елементи форматування мають бути доступні через кнопки; форматований текст — у окремому вікні на сторінці.


Програмний код:

// Форматування через document.execCommand
function fmt(cmd, val = null) {
  document.execCommand(cmd, false, val);
  document.getElementById('editorArea').focus();
}

// Зміна регістру через Selection API
function changeCase(type) {
  let sel   = window.getSelection();
  if (!sel.rangeCount) return;
  let range  = sel.getRangeAt(0);
  let text   = range.toString();
  if (!text) return;
  let result = type === 'upper' ? text.toUpperCase()
             : type === 'lower' ? text.toLowerCase()
             : text.replace(/\b\w/g, c => c.toUpperCase());
  range.deleteContents();
  range.insertNode(document.createTextNode(result));
}

function clearEditor() {
  if (confirm("Очистити весь текст?")) {
    document.getElementById('editorArea').innerHTML = '';
  }
}
