Завдання №6 — Функція makeArray()

Умова (варіант парний): Написати функцію makeArray(firstArray, secondArray, maxLength), яка створює новий масив з елементів обох масивів. Якщо довжина перевищує maxLength — повернути перші maxLength елементів. Інакше — повернути весь масив.


Програмний код:

function makeArray(firstArray, secondArray, maxLength) {
  let newArray = firstArray.concat(secondArray);
  if (newArray.length > maxLength) {
    return newArray.slice(0, maxLength);
  }
  return newArray;
}

// Виклики функції:
console.log( makeArray([1, 2, 3], [4, 5, 6], 5)  );  // [1, 2, 3, 4, 5]
console.log( makeArray([1, 2], [3, 4], 10)        );  // [1, 2, 3, 4]
console.log( makeArray([10], [20, 30, 40], 2)     );  // [10, 20]
