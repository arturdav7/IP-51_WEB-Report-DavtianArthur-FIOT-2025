Завдання №8 — Двовимірний масив

Умова (варіант парний): Є двовимірний масив додатніх і від'ємних чисел (оголосити, проініціалізувати генератором випадкових чисел). Вивести у Console. Зберегти перший і останній елементи масиву в окремі змінні. Вставити 25 після другого елемента масиву. Вивести у Console.


Програмний код:

// Генератор: числа від -10 до 10 (без 0)
function randNonZero() {
  let v;
  do { v = Math.floor(Math.random() * 21) - 10; } while (v === 0);
  return v;
}

// Двовимірний масив 3×4
let matrix = Array.from({ length: 3 }, () =>
  Array.from({ length: 4 }, randNonZero)
);

console.log("Двовимірний масив:");
matrix.forEach(row => console.log(row));

// Розгортаємо у плаский масив
let flat = matrix.flat();

let firstElement = flat[0];
let lastElement  = flat[flat.length - 1];
console.log(`Перший елемент: ${firstElement}`);
console.log(`Останній елемент: ${lastElement}`);

// Вставляємо 25 після другого елемента (індекс 2)
flat.splice(2, 0, 25);
console.log("Масив після вставки 25 після 2-го елемента:", flat);
