Завдання №8 — Форма логіна (submit + валідація)

Умова: Обробка форми за подією submit. Сторінка не перезавантажується. Якщо поля не заповнені — alert. Якщо заповнені — зібрати дані в об'єкт через elements, вивести в консоль і очистити форму методом reset().


Програмний код:

const form = document.querySelector('.login-form');

form.addEventListener('submit', (e) => {
  e.preventDefault();

  const email    = form.elements.email.value.trim();
  const password = form.elements.password.value.trim();

  if (!email || !password) {
    alert('All form fields must be filled in');
    return;
  }

  const userData = {
    email:    form.elements.email.value.trim(),
    password: form.elements.password.value.trim(),
  };

  console.log(userData);
  form.reset();
});
