Завдання №2 — Варіант 7: Клас Calculator (method chaining)

Умова: Розробити клас Calculator з методами number(value), add(value), subtract(value), multiply(value), divide(value), getResult(). Підтримка ланцюжкового виклику (method chaining). При діленні на 0 — кидати помилку.


Програмний код:

class Calculator {
  number(value) {
    this.result = value;
    return this;
  }

  getResult() {
    return this.result;
  }

  add(value) {
    this.result += value;
    return this;
  }

  subtract(value) {
    this.result -= value;
    return this;
  }

  multiply(value) {
    this.result *= value;
    return this;
  }

  divide(value) {
    if (value === 0) throw new Error("Ділення на нуль неможливе!");
    this.result /= value;
    return this;
  }
}

// Приклад використання:
const calc = new Calculator();

const result = calc
  .number(10)    // 10
  .add(5)        // 10 + 5 = 15
  .subtract(3)   // 15 - 3 = 12
  .multiply(4)   // 12 * 4 = 48
  .divide(2)     // 48 / 2 = 24
  .getResult();

console.log(result); // 24

// Перевірка ділення на 0:
try {
  new Calculator().number(10).divide(0).getResult();
} catch (e) {
  console.log("Помилка:", e.message);
}
