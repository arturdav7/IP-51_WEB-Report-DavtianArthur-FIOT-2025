Завдання №1 — Варіант 10: checkBrackets(str)

Умова: Написати функцію checkBrackets(str), яка приймає рядок JS коду і перевіряє правильність закриття дужок (), {}, []. Якщо рядок містить коректний код — повертає true, інакше — false.


Програмний код:

function checkBrackets(str) {
  const stack = [];
  const pairs = { ')': '(', '}': '{', ']': '[' };

  for (let char of str) {
    if ('({['.includes(char)) {
      stack.push(char);                           // відкриваюча — в стек
    } else if (')}]'.includes(char)) {
      if (stack.pop() !== pairs[char]) return false; // не співпадає
    }
  }

  return stack.length === 0; // стек порожній — всі дужки закриті
}

console.log(checkBrackets("function test() { return [1, 2]; }"));  // true
console.log(checkBrackets("function test( { return [1, 2]; }"));   // false
console.log(checkBrackets("const a = (b + c) * [d - e]"));         // true
console.log(checkBrackets("const x = {a: [1, 2}]"));               // false
